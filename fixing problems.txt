import serial
import time

angles = [0, 50, 100, 50, 0, -50, -100, -50, 0]

s = serial.Serial('/dev/serial0', 115200, timeout=0.1)

print("Waiting Arduino boot...")
time.sleep(2)

# =========================================================
# WAIT FOR RFID AUTH + READY
# =========================================================

authorized = False
ready = False

print("Waiting for STATUS:AUTHORIZED + STATUS:FREE ...")

while True:
    if s.in_waiting:
        line = s.readline().decode(errors='ignore').strip()

        if line:
            print("RX:", line)

        if line == "STATUS:AUTHORIZED":
            authorized = True

        if authorized and line == "STATUS:FREE":
            ready = True
            break

print("\nArduino READY\n")

# =========================================================
# ENTER ASSIST MODE
# =========================================================

print("TX: CMD:ASSIST")
s.write(b'CMD:ASSIST\n')

assist_ok = False
t0 = time.time()

while time.time() - t0 < 3:
    if s.in_waiting:
        line = s.readline().decode(errors='ignore').strip()

        if line:
            print("RX:", line)

        if line == "STATUS:ASSIST":
            assist_ok = True
            break

if not assist_ok:
    print("Failed to enter ASSIST mode")
    s.close()
    raise SystemExit

print("\nASSIST MODE ACTIVE\n")

# =========================================================
# SEND ANGLES
# =========================================================

for angle in angles:

    cmd = f'CMD:ANGLE:{angle}\n'

    print("TX:", cmd.strip())

    s.write(cmd.encode())

    t0 = time.time()

    while time.time() - t0 < 3:

        if s.in_waiting:

            line = s.readline().decode(errors='ignore').strip()

            if line:
                print("RX:", line)

            # stop reading when movement cycle ends
            if line in [
                "STATUS:REACHED",
                "STATUS:AT_TARGET",
                "STATUS:LOCKED_LEFT",
                "STATUS:LOCKED_RIGHT",
                "STATUS:OUT_OF_RANGE"
            ]:
                break

    time.sleep(1)

s.close()

print("\nDONE")
